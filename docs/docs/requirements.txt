# Documentation-build toolchain for the forgeloop docs site.
#
# Install the forgeloop package first, then these:
#
#   python -m pip install -e ../..          # the forgeloop package (repo root)
#   python -m pip install -r requirements.txt
#
# Build:  cd forgeloop-docs/docs && make html   (output in docs/_build/html)
#
# The API reference imports forgeloop with torch / knowlytix / transformers /
# peft / anthropic / huggingface_hub / pandas mocked (see autodoc_mock_imports
# in conf.py), so a plain docs environment does not need those installed. numpy,
# pydantic and pyyaml are real, light dependencies pulled in by the package.

sphinx>=8.0
sphinx-gallery>=0.19
sphinx-rtd-theme>=3.0
# napoleon (Google/NumPy docstring parsing) ships with Sphinx; no extra dep.
